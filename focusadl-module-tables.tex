\ProvidesFile{focusadl-module-tables.tex}[2026-08-06 v1.8
   Table support for the FOCUS Adelaide Document Class (DCP)]

\RequirePackage{tabularray}
\UseTblrLibrary{booktabs}

\SetTblrTemplate{caption}{empty}
\SetTblrTemplate{conthead}{empty}
\SetTblrTemplate{contfoot}{empty}
\SetTblrTemplate{caption-tag}{empty}
\SetTblrTemplate{caption-sep}{empty}
\DefTblrTemplate{contfoot-text}{default}{}
\DefTblrTemplate{conthead-text}{default}{}

\SetTblrInner{
  row{1} = {font=\fontseries{sb}\selectfont},
}

\NewDocumentEnvironment{FOCUSAdlTable}{m+b}{
  \begin{tblr}{#1}
    #2
  \end{tblr}%
  \ignorespaces
}{}

\NewDocumentEnvironment{FOCUSAdlLongTable}{m+b}{
  \begin{longtblr}{#1}
    #2
  \end{longtblr}%
  \ignorespaces
}{}

\endinput
