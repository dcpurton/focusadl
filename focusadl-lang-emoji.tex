% Hide warnings about missing shapes for Noto Color Emoji font
\ExplSyntaxOn
\makeatletter
\bool_new:N \l__focusadl_show_font_warning_bool
\cs_set_protected:Npn \@font@warning #1
  {
    \bool_set_true:N \l__focusadl_show_font_warning_bool
    \tl_if_in:nnTF {#1} { \@gobbletwo }
      {
        \bool_set_false:N \l__focusadl_show_font_warning_bool
      }
      {
        \exp_args:Ne \str_if_in:nnT {#1} { NotoColorEmoji }
          {
            \bool_set_false:N \l__focusadl_show_font_warning_bool
          }
      }
    \bool_if:NT \l__focusadl_show_font_warning_bool
      {
        \GenericWarning { (Font) \@spaces \@spaces \@spaces \space \space }
                        { LaTeX ~ Font ~ Warning: ~ #1 }
      }
  }
\makeatother
\ExplSyntaxOff

\babelprovide
  [
    , onchar = fonts
    , identification/tag.bcp47 = en
    , characters/ranges = 1F300..1F9FF 2600..26FF 2700..27BF
  ]
  {emoji}

\babelfont[emoji]{rm}{Noto Color Emoji}
