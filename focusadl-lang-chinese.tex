\RequirePackage{luacode}

\begin{luacode}
function fix_cjk_quote_kerning(s)
  s = s:gsub ("(\\vs%b{})(“)", "%1\\kern-.55em%2")
  s = s:gsub ("(\\vs%b{})(‘)", "%1\\kern-.65em%2")
  s = s:gsub ("(\\ch%b{})(“)", "%1\\kern-.6em%2")
  s = s:gsub ("(\\ch%b{})(‘)", "%1\\kern-.7em%2")
  return s
end
\end{luacode}

\newcommand{\FOCUSAdlCJKQuoteKerningOn}{%
  \directlua{%
    luatexbase.add_to_callback(
      "process_input_buffer", fix_cjk_quote_kerning, "fix_cjk_quote_kerning")}}
\newcommand{\FOCUSAdlCJKQuoteKerningOff}{%
  \directlua{%
    luatexbase.remove_from_callback(
      "process_input_buffer", "fix_cjk_quote_kerning")}}
